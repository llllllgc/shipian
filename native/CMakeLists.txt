cmake_minimum_required(VERSION 3.14)
project(shipian_raw_converter LANGUAGES CXX)
set(LIBRAW_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/LibRaw-0.22.2")
file(GLOB_RECURSE LIBRAW_SOURCES CONFIGURE_DEPENDS "${LIBRAW_DIR}/src/*.cpp")
list(FILTER LIBRAW_SOURCES EXCLUDE REGEX "_ph\\.cpp$")
add_executable(shipian_raw_converter raw_converter.cpp ${LIBRAW_SOURCES})
target_compile_features(shipian_raw_converter PRIVATE cxx_std_17)
target_include_directories(shipian_raw_converter PRIVATE "${LIBRAW_DIR}")
target_compile_definitions(shipian_raw_converter PRIVATE LIBRAW_NODLL LIBRAW_NOTHREADS NOMINMAX)
target_compile_options(shipian_raw_converter PRIVATE /EHsc /w /utf-8)
target_link_libraries(shipian_raw_converter PRIVATE windowscodecs ole32 oleaut32 ws2_32)
